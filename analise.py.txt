import pandas as pd

# Dados das estatísticas de roubo de veículos no RJ - 1º Trimestre 2025
# Fonte: Instituto de Segurança Pública (ISP-RJ)
data = {
    'Mes': ['Janeiro', 'Fevereiro', 'Março'],
    'Roubos_Veiculos': [2500, 2591, 1483],
    'Veiculos_Recuperados': [1700, 1800, 1541]
}

df = pd.DataFrame(data)

# Cálculo do total do trimestre
total_trimestre = df['Roubos_Veiculos'].sum()
recuperados_total = df['Veiculos_Recuperados'].sum()

print("Estatísticas de Roubo de Veículos - RJ (Q1 2025)")
print(df.to_markdown(index=False))
print(f"\nTotal de Roubos no Trimestre: {total_trimestre}")
print(f"Total de Veículos Recuperados: {recuperados_total} (Aumento de 29% vs 2024)")
